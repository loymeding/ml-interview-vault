# 📚 Индекс источников

> [!abstract] Назначение
> Все карточки источников, по которым ведётся изучение или которые уже пройдены. Для вишлиста — [[📖 Очередь чтения]].

---

## Все источники

```dataviewjs
const statusOrder = { "читаю": 0, "не начато": 1, "пауза": 2, "прочитано": 3 };
const priorityEmoji = { "высокий": "🔴", "средний": "🟡", "низкий": "🟢" };

const pages = dv.pages('"Литература"')
  .where(p => p["статус"] && p.file.name !== "📚 Индекс источников" && p.file.name !== "Шаблон источника")
  .sort(p => [statusOrder[p["статус"]] ?? 9, p["приоритет"]], "asc");

dv.table(
  ["Источник", "Тип", "Авторы", "Статус", "Приоритет", "Топик"],
  pages.map(p => [
    p.file.link,
    p["тип"] ?? "—",
    Array.isArray(p["авторы"]) ? p["авторы"].join(", ") : (p["авторы"] ?? "—"),
    p["статус"] ?? "—",
    (priorityEmoji[p["приоритет"]] ?? "") + " " + (p["приоритет"] ?? "—"),
    p["топик"] ? p["топик"].join(", ") : "—"
  ])
);
```

## По статусам

```dataviewjs
const sections = ["читаю", "не начато", "пауза", "прочитано"];
const icons = { "читаю": "📖", "не начато": "⏳", "пауза": "⏸️", "прочитано": "✅" };

for (const status of sections) {
  const pages = dv.pages('"Литература"')
    .where(p => p["статус"] === status);
  if (pages.length === 0) continue;
  dv.header(3, `${icons[status]} ${status[0].toUpperCase() + status.slice(1)} (${pages.length})`);
  dv.list(pages.map(p => p.file.link));
}
```
