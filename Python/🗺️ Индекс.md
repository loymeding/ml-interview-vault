## 📊 Статистика хранилища

```dataview
TABLE WITHOUT ID
  length(rows) as "📚 Всего",
  length(filter(rows, (r) => r.статус = "готово")) as "✅ Готово",
  length(filter(rows, (r) => r.статус = "черновик")) as "✏️ Черновик",
  length(filter(rows, (r) => r.статус = "требует доработки")) as "⚠️ Доработки",
  length(filter(rows, (r) => r.сложность = "высокая")) as "🔴 Сложных",
  length(filter(rows, (r) => r.тип = "сравнение")) as "🆚 Сравнений"
FROM "Python"
WHERE статус
  AND file.folder != "Python"
  AND file.folder != "Python/Шаблоны"
GROUP BY true
```

## 🧱 По типам заметок

```dataview
TABLE WITHOUT ID
  key as "Тип",
  length(rows) as "Количество",
  join(map(rows, (r) => link(r.file.path, r.file.name)), ", ") as "Заметки"
FROM "Python"
WHERE статус
  AND file.name != "Шаблон новой темы"
  AND file.name != "Шаблон сравнительной темы"
  AND file.folder != "Python"
  AND file.folder != "Python/Шаблоны"
GROUP BY default(тип, "не задан")
SORT key ASC
```

## 📂 По разделам

```dataview
TABLE WITHOUT ID
  file.folder as "Раздел",
  rows.file.link as "Заметки"
FROM "Python"
WHERE статус
  AND file.folder != "Python"
  AND file.folder != "Python/Шаблоны"
GROUP BY file.folder
SORT file.folder ASC
```

## 📘 Теория для повторения

```dataview
TABLE WITHOUT ID
  file.link as "Карточка",
  уровень as "Уровень",
  сложность as "Сложность",
  готовность as "Готовность"
FROM "Python"
WHERE статус
  AND (тип = "theory" OR тип = "теория и практика")
  AND file.folder != "Python"
  AND file.folder != "Python/Шаблоны"
SORT сложность DESC, file.name ASC
```

## 🧪 Coding Tasks для лайвкодинга

```dataview
TABLE WITHOUT ID
  file.link as "Набор задач",
  topic as "Тема",
  сложность as "Сложность",
  card_ref as "Теория"
FROM "Python/Coding Tasks"
WHERE тип = "coding-task"
SORT сложность DESC, file.name ASC
```
