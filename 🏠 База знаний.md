const exclude = p =>
  !p.file.path.startsWith(".") &&
  !p.file.folder.startsWith(".trash") &&
  !p.file.folder.startsWith(".obsidian") &&

const all = dv.pages().where(exclude);
const now = Date.now();

## ⚙️ Алгоритмы и кодинг

### LeetCode: общий прогресс

```dataview
TABLE WITHOUT ID
  сложность AS "Сложность",
  length(rows) AS "Всего",
  length(filter(rows, (r) => r.статус = "решено")) AS "✅ Решено",
  length(rows) - length(filter(rows, (r) => r.статус = "решено")) AS "⬜ Осталось",
  round(length(filter(rows, (r) => r.статус = "решено")) / length(rows) * 100) + "%" AS "Прогресс"
FROM "LeetCode/Задачи"
WHERE номер
GROUP BY сложность
SORT сложность ASC
```

### Interview problems

```dataview
TABLE WITHOUT ID
  источник AS "Источник",
  length(rows) AS "Всего",
  length(filter(rows, (r) => r.статус = "решено")) AS "✅ Решено",
  length(rows) - length(filter(rows, (r) => r.статус = "решено")) AS "⬜ Осталось"
FROM "LeetCode/Interview Problems"
WHERE интервью = true
GROUP BY источник
SORT источник ASC
```

## 🎙️ Интервью и обратная связь

```dataview
TABLE WITHOUT ID
  file.link AS "Разбор",
  interview_date AS "Дата",
  role AS "Роль",
  questions AS "Вопросов",
  avg_score AS "Средний score",
  red_count AS "🔴 1-2/5",
  yellow_count AS "🟡 3/5",
  green_count AS "🟢 4-5/5",
  high_priority_count AS "🔥 High"
FROM "Interview Reviews/Разборы"
WHERE contains(tags, "interview-review")
SORT interview_date DESC
```

> [!danger] 🔥 Принцип
> Главная метрика интервью — не “как прошло”, а сколько слабых мест превращено в уверенные ответы. После каждого собеседования сюда должны попадать вопросы с `priority:: high`, а затем уходить из очереди после повторения.

## 🆕 Недавно измененные карточки

```dataview
TABLE WITHOUT ID
  file.link AS "Карточка",
  file.folder AS "Раздел",
  dateformat(file.mtime, "yyyy-MM-dd HH:mm") AS "Изменено"
FROM ""
WHERE !startswith(file.folder, ".")
  AND !startswith(file.folder, ".trash")
  AND file.name != "🏠 База знаний"
SORT file.mtime DESC
LIMIT 15
```
